$gateResults = [System.Collections.Generic.List[object]]::new()
function Invoke-FinalGate([string]$Name, [string]$Executable, [string[]]$Arguments) {
  $gateLog = Join-Path $PSScriptRoot ('post-review-final-' + $Name + '.log')
  $started = Get-Date
  Write-Output ('Starting ' + $Name + ': ' + $Executable + ' ' + ($Arguments -join ' '))
  & $Executable @Arguments *> $gateLog
  $gateExit = $LASTEXITCODE
  $gateResults.Add([pscustomobject]@{ name=$Name; command=($Executable+' '+($Arguments -join ' ')); exitCode=$gateExit;
    startedAt=$started.ToString('o'); durationSeconds=[Math]::Round(((Get-Date)-$started).TotalSeconds,2);
    sourceHead=$sourceHead; logSha256=(Get-FileHash -Algorithm SHA256 -LiteralPath $gateLog).Hash; retainedPriorRun=$false })
  $gateResults | ConvertTo-Json -Depth 4 | Set-Content -LiteralPath (Join-Path $PSScriptRoot 'post-review-final-gates.json')
  Get-Content -LiteralPath $gateLog -Tail 12
  if ($gateExit -ne 0) { Write-Output ('FAILED ' + $Name + ': ' + $gateExit); exit $gateExit }
  Write-Output ('PASSED ' + $Name + ': exit 0')
}
