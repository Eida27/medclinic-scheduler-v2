function Invoke-FinalGate([string]$Name, [string]$Executable, [string[]]$Arguments) {
  $gateLog = Join-Path $PSScriptRoot ('final-' + $Name + '.log')
  $gateStarted = Get-Date
  Write-Output ('Starting ' + $Name + ': ' + $Executable + ' ' + ($Arguments -join ' '))
  & $Executable @Arguments *> $gateLog
  $gateExit = $LASTEXITCODE
  $gateResults.Add([pscustomobject]@{ name=$Name; command=($Executable+' '+($Arguments -join ' ')); exitCode=$gateExit; startedAt=$gateStarted.ToString('o'); durationSeconds=[Math]::Round(((Get-Date)-$gateStarted).TotalSeconds,2); sourceHead=$actualSourceHead; logSha256=(Get-FileHash -Algorithm SHA256 -LiteralPath $gateLog).Hash })
  $gateResults | ConvertTo-Json -Depth 4 | Set-Content (Join-Path $PSScriptRoot 'final-gates.json')
  Get-Content -LiteralPath $gateLog -Tail 12
  if ($gateExit -ne 0) { Write-Output ('FAILED ' + $Name + ': ' + $gateExit); exit $gateExit }
  Write-Output ('PASSED ' + $Name + ': exit 0')
}
